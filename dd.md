        function handleCalendar() {
            const today = new Date();
            const year = today.getFullYear();
            const month = today.getMonth();

            const minDate = new Date(year, month, 1);
            const maxDate = new Date(year, month + 1, 0);

        function markSundays(selectedDates, dateStr, instance) {
            // 3가지 인자는 flatpickr의 통신규약 (변수를 직접 사용하지 않더라도 인자로 받아야 함);
            const days = instance.calendarContainer.querySelectorAll(".flatpickr-day");

            days.forEach(function (day) {
                // 전월, 익월을 달력에 표기하지 않고, 요일 계산에도 반영
                if (
                    !day.classList.contains("prevMonthDay") &&
                    !day.classList.contains("nextMonthDay")
                ) {
                    var label = day.ariaLabel;
                    var match = label.match(/(\d+)월 (\d+), (\d{4})/);

                    if (match) {
                        var month = match[1];
                        var dayNum = match[2];
                        var year = match[3];

                        var date = new Date(`${year}-${month.padStart(2, "0")}-${dayNum.padStart(2, "0")}`);

                        // 일 ~ 토 = 0 ~ 6
                        if (date.getDay() === 0) {
                            day.style.color = "red";
                        } else {
                            day.style.color = ""; // 일요일이 아니면 색상 초기화
                        }
                    }
                }
            });
        }

        const CherryBlossom = (() => {

            // 속성 설정. 속도, 크기, 생성주기 등
            const defaults = {
                speed: 5,
                maxSize: 8,
                minSize: 6,
                newOn: 400,
                container: '.cherry_blossom'
            };

            // 재할당을 위한 let
            let container = null;
            let wrapH = 0;
            let wrapW = 0;
            let petalTemplate = null;
            let randomSwayAnims = [];

            // 랜덤 회전 값 생성
            const getRandomRotate = () => {
                const rotateX = 360;
                const rotateY = Math.random() * 60 - 30;
                const rotateZ = Math.random() * 120 - 30;
                const translateX = Math.random() * 10 - 5;
                const translateY = Math.random() * 10 - 10;
                const translateZ = Math.random() * 15;
                return `rotateX(${rotateX}deg) rotateY(${rotateY}deg) rotateZ(${rotateZ}deg) translateX(${translateX}px) translateY(${translateY}px) translateZ(${translateZ}px)`;
            };

            // 랜덤 회전 값 배열 만들기
            const createRandomSwayAnimations = () => {
                return Array(10).fill().map(getRandomRotate);
            };

            // 벚꽃 템플릿 생성
            const createPetalTemplate = () => {
                const template = document.createElement('span');
                template.className = 'petal';
                return template;
            };

            // 공간 크기
            const updateDimensions = () => {
                wrapH = container.clientHeight;
                wrapW = container.clientWidth;
            };

            // 회전 애니메이션
            // generatePetal 함수에서 넘겨받는 petal 값을 재귀적으로 할당
            const applySwayAnim = (element) => {
                const randomSway = randomSwayAnims[Math.floor(Math.random() * randomSwayAnims.length)];
                element.style.transform = randomSway;

                setTimeout(() => {
                    applySwayAnim(element);
                }, 1000);
            };

            // 벚꽃 생성
            const generatePetal = () => {
                const petal = petalTemplate.cloneNode(true);
                const size = Math.floor(Math.random() * (defaults.maxSize - defaults.minSize + 1)) + defaults.minSize;
                // 시작위치를 오른쪽 상단으로 이동
                const startPosLeft = Math.random() * wrapW;
                const fallTime = (wrapH * 0.1 + Math.random() * 5) / defaults.speed;
                const horizontalOffset = Math.random() * 2 - 1; // 좌우 흔들림

                petal.addEventListener('animationend', () => {
                    petal.remove();
                });

                petal.style.cssText = `
                    width: ${size}px;
                    height: ${size}px;
                    left: ${startPosLeft}px;
                    position: absolute;
                    animation: sacura-fall ${fallTime}s linear;
                `;

                container.appendChild(petal);

                // 위치 업데이트
                const updatePos = () => {
                    // 값 계산을 위해 숫자만 추출
                    const currentLeft = parseFloat(petal.style.left);
                    petal.style.left = `${currentLeft + horizontalOffset}px`;
                    requestAnimationFrame(updatePos);
                };

                updatePos();

                // 회전 애니메이션
                applySwayAnim(petal);
            };

            // 애니메이션 저장용 변수
            let animationId = null;

            // 애니메이션 시작
            const startAnimation = () => {
                randomSwayAnims = createRandomSwayAnimations();

                const animate = () => {
                    generatePetal();
                    animationId = setTimeout(animate, defaults.newOn);
                };

                requestAnimationFrame(animate);
            };

            const stop = () => {
                // console.log('animationId', animationId);
                // console.log('container', container);

                if (animationId) {
                    clearTimeout(animationId);
                    animationId = null;
                }
                // 기존 벚꽃 남아있다면 제거
                if (container) {
                    container.innerHTML = '';
                }
            };

            const init = () => {
                // 컨테이너 요소 선택
                container = document.querySelector(defaults.container);
                if (!container) {
                    console.log('container 요소가 존재하지 않습니다');
                    return;
                }

                // 초기화
                updateDimensions();
                petalTemplate = createPetalTemplate();
                startAnimation();
            };

            return {
                init,
                stop
            };
        })();

        function handleCommonPopup() {
            const overlay = document.querySelectorAll('.modal, .popup');
            const buttons = document.querySelectorAll('.modal .component_button, .popup .component_button');

            // 배경영역 클릭 시, 팝업(모달) 감추기
            overlay.forEach(overlay => {
                overlay.addEventListener('click', (e) => {
                    // 오버레이 배경을 클릭한 경우
                    if (e.target === overlay) {
                        overlay.classList.remove('visible');
                    };
                });
            });

            buttons.forEach(button => {
                button.addEventListener('click', (e) => {
                    // 태그 기본속성 차단
                    e.preventDefault();

                    // 임의. 모든 버튼은 팝업(모달)을 감춤.
                    const visiblePopup = document.querySelector('.modal.visible, .popup.visible');
                    if (visiblePopup) {
                        visiblePopup.classList.remove('visible');
                    }
                });
            });

            // 팝업을 보여주는 버튼 class
            const openButtons = document.querySelectorAll('.open-popup');

            openButtons.forEach(button => {
                button.addEventListener('click', (e) => {
                    e.preventDefault();

                    const dataId = button.dataset.popup;
                    // 버튼과 팝업(모달)의 dataset 비교 (같을 경우 매칭)
                    const targetPopup = document.querySelector(`.modal[data-popup="${dataId}"], .popup[data-popup="${dataId}"]`);

                    if (targetPopup) {
                        targetPopup.classList.add('visible');

                        console.log('targetPopup', targetPopup);
                    }
                });
            });

        };

        function handleExpandedItem() {
            const button = document.querySelector('.component_button.has-item');
            const toggleItem = document.querySelector('.cont-row.message .toggle-item');

            button.addEventListener('click', function (e) {
                e.preventDefault();
                const text = button.querySelector('.text');
                const isExpanded = button.getAttribute('aria-expanded') === 'true';

                console.log('isExpanded', isExpanded);

                if (isExpanded) {
                    text.innerText = '메시지 남기기';
                    toggleItem.style.display = 'none';
                    button.setAttribute('aria-expanded', 'false');
                } else {
                    text.innerText = '등록하기';
                    toggleItem.style.display = 'block';
                    button.setAttribute('aria-expanded', 'true');
                }
            });
        };

        function handleToggle() {
            const buttons = document.querySelectorAll('.component_button.mute, .component_button.play');

            buttons.forEach(button => {
                button.addEventListener('click', function (e) {
                    e.preventDefault();

                    const isplayed = button.classList.contains('play');

                    buttons.forEach(b => b.classList.remove('visible'));

                    if (isplayed) {
                        buttons.forEach(b => {
                            if (!b.classList.contains('play')) {
                                b.classList.add('visible');
                            }
                        })
                    } else {
                        buttons.forEach(b => {
                            if (b.classList.contains('play')) {
                                b.classList.add('visible');
                            }
                        })
                    }
                })
            })
        };
